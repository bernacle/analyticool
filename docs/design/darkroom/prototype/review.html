select month,
  sum(gross) - sum(refunds) - sum(fees) as net
from (
  select date_trunc('month', c.created),
    (c.amount_captured - c.amount_tax)
      * bt.exchange_rate, 0, bt.fee
  from stripe.charges c
  join stripe.balance_transactions bt
    on bt.id = c.balance_transaction
  where c.status = 'succeeded'
    and c.livemode
  union all
  select date_trunc('month', r.created),
    0, r.amount * rbt.exchange_rate, 0
  from stripe.refunds r
  join stripe.balance_transactions rbt
    on rbt.id = r.balance_transaction
) t (month, gross, refunds, fees)
group by 1 order by 1;
